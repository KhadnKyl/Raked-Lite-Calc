function drawShape(W,LH,RH,S,GW,GLH,GRH){
  const svg=document.getElementById('shape');
  const pad=62, vw=720, vh=340;
  const maxH=Math.max(LH,RH);
  const sx=(vw-2*pad)/W;
  const sy=(vh-2*pad)/maxH;
  const sc=Math.min(sx,sy);
  const x0=(vw-W*sc)/2;
  const yBase=vh-pad/2;
  const p=(x,y)=>`${(x0+x*sc).toFixed(1)},${(yBase-y*sc).toFixed(1)}`;
  const framePts=[p(0,0),p(W,0),p(W,RH),p(0,LH)].join(' ');
  let glassPts="";
  if(GW>0&&GLH>0&&GRH>0){
    glassPts=[p(S,S),p(W-S,S),p(W-S,S+GRH),p(S,S+GLH)].join(' ');
  }

  const highLabel = LH>RH ? "LONG" : LH<RH ? "SHORT" : "";
  const rightLabel = LH>RH ? "SHORT" : LH<RH ? "LONG" : "";

  svg.innerHTML=`
    <polygon points="${framePts}" fill="rgba(63,182,211,.08)" stroke="#7fd3e4" stroke-width="4" stroke-linejoin="round"/>
    ${glassPts?`<polygon points="${glassPts}" fill="rgba(243,197,107,.08)" stroke="#f3c56b" stroke-width="3" stroke-dasharray="8 5" stroke-linejoin="round"/>`:''}
    <text x="${x0+W*sc/2}" y="${yBase+30}" text-anchor="middle" fill="#a9bbc6" font-size="15">${fmt(W,0)} mm</text>
    <text x="${x0-14}" y="${yBase-LH*sc/2}" text-anchor="end" fill="#a9bbc6" font-size="14">LH ${fmt(LH,0)}</text>
    <text x="${x0+W*sc+14}" y="${yBase-RH*sc/2}" text-anchor="start" fill="#a9bbc6" font-size="14">RH ${fmt(RH,0)}</text>
    ${LH>RH
      ? `<text x="${x0+10}" y="${Math.max(28,yBase-LH*sc-12)}" text-anchor="start" fill="#8edff0" font-size="12">${highLabel}</text>
         <text x="${x0+W*sc+14}" y="${Math.max(28,yBase-RH*sc-12)}" text-anchor="start" fill="#8edff0" font-size="12">${rightLabel}</text>`
      : LH<RH
      ? `<text x="${x0-14}" y="${Math.max(28,yBase-LH*sc-12)}" text-anchor="end" fill="#8edff0" font-size="12">${highLabel}</text>
         <text x="${x0+W*sc-10}" y="${Math.max(28,yBase-RH*sc-12)}" text-anchor="end" fill="#8edff0" font-size="12">${rightLabel}</text>`
      : ""}
    <text x="${x0+W*sc/2}" y="26" text-anchor="middle" fill="#d9e5eb" font-size="13">Overall sizes — preview only</text>
  `;
}

function drawJambSketch(svgId, capId, side, length, angle, topIsLong){
  const svg=document.getElementById(svgId);
  const cap=document.getElementById(capId);

  function extLine(A, B, ext){
    const dx=B[0]-A[0], dy=B[1]-A[1];
    const len=Math.hypot(dx,dy) || 1;
    const ux=dx/len, uy=dy/len;
    return [A[0]-ux*ext, A[1]-uy*ext, B[0]+ux*ext, B[1]+uy*ext];
  }

  const yTop=58, yBot=126, rightX=288;
  const leftTopX = topIsLong ? 50 : 104;
  const leftBotX = topIsLong ? 104 : 50;
  const pointName = topIsLong ? "LONG POINT" : "SHORT POINT";

  const A=[leftTopX,yTop], B=[leftBotX,yBot];
  const E=extLine(A,B,28);

  svg.innerHTML = `
    <rect x="0" y="0" width="360" height="180" rx="10" fill="#101b23"/>

    <polygon points="${A[0]},${A[1]} ${rightX},${yTop} ${rightX},${yBot} ${B[0]},${B[1]}"
      fill="none" stroke="#eef7fb" stroke-width="4" stroke-linejoin="round"/>

    <line x1="${E[0]}" y1="${E[1]}" x2="${E[2]}" y2="${E[3]}"
      stroke="#7a93a2" stroke-width="2" stroke-dasharray="6 4"/>

    <line x1="${A[0]}" y1="34" x2="${rightX}" y2="34"
      stroke="#8edff0" stroke-width="2"/>
    <line x1="${A[0]}" y1="28" x2="${A[0]}" y2="40"
      stroke="#8edff0" stroke-width="2"/>
    <line x1="${rightX}" y1="28" x2="${rightX}" y2="40"
      stroke="#8edff0" stroke-width="2"/>
    <text x="${(A[0]+rightX)/2}" y="24" text-anchor="middle"
      fill="#8edff0" font-size="17" font-weight="700">${fmt(length)} mm</text>

    <circle cx="${A[0]}" cy="${A[1]}" r="4" fill="#8edff0"/>
    <line x1="${A[0]}" y1="${A[1]+4}" x2="${A[0]}" y2="138" stroke="#8edff0" stroke-width="1.5"/>

    <text x="${Math.min(A[0],B[0])-14}" y="97" text-anchor="middle"
      fill="#f3c56b" font-size="12">${fmt(angle,1)}°</text>

    <text x="${A[0]}" y="151" text-anchor="middle" fill="#e5f4f9" font-size="11">${pointName}</text>
    <text x="176" y="166" text-anchor="middle" fill="#e5f4f9" font-size="11">BOTTOM SQUARE / 0°</text>
  `;

  cap.innerHTML = `<b>${side} jamb:</b> ${fmt(length)} mm from square bottom to top <b>${pointName.toLowerCase()}</b>. Dashed line shows the blade / mitre line.`;
}


function renderCutSketches(angle, signedRise, headCut, sillCut, LH, RH){
  const isRHRake = signedRise < 0;
  drawHeadSketch(angle, signedRise, headCut);
  drawSillSketch(sillCut);
  if(signedRise === 0){
    drawJambSketch('lhSketch','lhSketchCap','LH',LH,0,true);
    drawJambSketch('rhSketch','rhSketchCap','RH',RH,0,true);
  } else if(isRHRake){
    drawJambSketch('lhSketch','lhSketchCap','LH',LH,angle,true);
    drawJambSketch('rhSketch','rhSketchCap','RH',RH,angle,false);
  } else {
    drawJambSketch('lhSketch','lhSketchCap','LH',LH,angle,false);
    drawJambSketch('rhSketch','rhSketchCap','RH',RH,angle,true);
  }
}
